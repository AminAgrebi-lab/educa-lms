{% endblock %} {% block include_js %} {% endblock %} {% block domready %} // WebSocket endpoint for THIS course room (id injected by Django) const url = 'ws://' + window.location.host + '/ws/chat/room/' + {{ course.id }} + '/'; const chatSocket = new WebSocket(url); // Append incoming messages to the chat area chatSocket.onmessage = function (event) { const data = JSON.parse(event.data); const message = document.createElement('div'); message.className = 'message'; message.textContent = data.message; document.querySelector('#chat').appendChild(message); }; chatSocket.onclose = function (event) { console.error('Chat socket closed unexpectedly'); }; const input = document.querySelector('#chat-message-input'); const submit = document.querySelector('#chat-message-submit'); // The Send button listener (this was the missing piece!) submit.onclick = function (event) { const message = input.value; chatSocket.send(JSON.stringify({'message': message})); input.value = ''; }; input.focus(); input.onkeyup = function (event) { if (event.key === 'Enter') { submit.click(); } }; {% endblock %}
